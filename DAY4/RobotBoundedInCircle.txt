class Solution {
    public boolean isRobotBounded(String instructions) {
        int x=0,y=0;
        int dir=0; //0-N 1-E 2-S 3-W
        int[][] move={{0,1},  //North
                      {1,0},  //East
                      {0,-1}, //South
                      {-1,0}};//West
        for(char ch:instructions.toCharArray()){
            if(ch=='G'){
                x+=move[dir][0];
                y+=move[dir][1];
            }
            else if(ch=='L') dir=(dir+3)%4;
            else if(ch=='R') dir=(dir+1)%4;
        }
        return (x==0 && y==0) || dir!=0;
    }
}
