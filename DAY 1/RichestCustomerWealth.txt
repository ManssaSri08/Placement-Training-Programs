class Solution {
    public int maximumWealth(int[][] accounts) {
        int rich=0;
        for(int r=0;r<accounts.length;r++){
            int sum=0;
            for(int c=0;c<accounts[0].length;c++){
                sum+=accounts[r][c];
            }
            if(sum>rich) rich=sum;
        }
        return rich;
    }
}
